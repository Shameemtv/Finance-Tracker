const bytesToBase64 = (bytes) => {
let binary = '';
const arr = new Uint8Array(bytes);
for (let i = 0; i < arr.byteLength; i++) {
binary += String.fromCharCode(arr[i]);
}
return btoa(binary);
};
const base64ToBytes = (b64) => {
const binary = atob(b64);
const bytes = new Uint8Array(binary.length);
for (let i = 0; i < binary.length; i++) {
bytes[i] = binary.charCodeAt(i);
}
return bytes;
};

h1 = Math.imul(h1 ^ ch, 2654435761);

const getWeekBounds = (offset = 0) => {
const now = new Date();
const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
const day = today.getDay();
const diffToMonday = (day === 0 ? -6 : 1) - day;
const startOfWeek = new Date(today);
startOfWeek.setDate(today.getDate() + diffToMonday + (offset * 7));
const endOfWeek = new Date(startOfWeek);
endOfWeek.setDate(startOfWeek.getDate() + 6);
return { startOfWeek, endOfWeek };
};
const isWithinSelectedPeriod = (dateStr, period) => {
const [y, m, d] = dateStr.split('-').map(Number);
const txDate = new Date(y, m - 1, d);
if (period === 'weekly') {
const { startOfWeek, endOfWeek } = getWeekBounds(selectedWeekOffset);
return txDate >= startOfWeek && txDate <= endOfWeek;
}
if (period === 'monthly') return y === selectedYear && (m - 1) === selectedMonth;
if (period === 'yearly') return y === selectedYear;
return true;
};

try {
sourceData = await decryptObject({ iv: parsed.iv, ciphertext: parsed.ciphertext },
activeCryptoKey);
} catch (decErr) {
showStatus('Cannot decrypt: backup belongs to a different password', true);
return;
}
}
const rawList = Array.isArray(sourceData) ? sourceData : sourceData.transactions;
if (!Array.isArray(rawList)) throw new Error('Invalid format');
if (!Array.isArray(sourceData) && Array.isArray(sourceData.customCategories)) {
customCategories = sourceData.customCategories
.filter(c => c && typeof c.name === 'string' && c.name.trim())
.map(c => ({
name: c.name.trim(),
type: c.type === 'income' ? 'income' : 'expense'
}));

...BASE_DAILY_CATEGORIES.map(c => c.name.toLowerCase()),
...BORROW_CATEGORIES.map(c => c.toLowerCase())
];
const sanitized = rawList
.filter(item => item && Number(item.amount) > 0 && typeof item.date === 'string')
.map((item, idx) => {
const catName = String(item.category || 'Other').trim();
const txType = item.type === 'income' ? 'income' : 'expense';
const isBuiltIn = builtInNamesLower.includes(catName.toLowerCase());
const isAlreadyCustom = customCategories.some(c => c.name.toLowerCase() ===
catName.toLowerCase());
if (!isBuiltIn && !isAlreadyCustom && catName) {
customCategories.push({ name: catName, type: txType });
}
return {
id: Number(item.id) || (Date.now() + idx),
type: txType,
category: catName || 'Other',
person: String(item.person || '').trim(),
amount: parseFloat(Number(item.amount).toFixed(2)),
reason: String(item.reason || catName || 'Other').trim(),
date: item.date
};
});

sourceData.monthlyBudget >= 0) {
monthlyBudget = sourceData.monthlyBudget;
}
editingId = null;
await persistVault();
render();
showStatus(`Restored ${sanitized.length} entries`);
} catch (err) {
showStatus('Invalid backup file', true);
}
